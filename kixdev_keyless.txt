local function loadScript()
	local ok, err = pcall(function()
		loadstring(game:HttpGet(SCRIPT_URL, true))()
	end)
	if not ok then
		warn("[KixDev] Failed to load main script: " .. tostring(err))
	end
end

local function saveKeyCache()
	if typeof(writefile) == "function" then
		pcall(function()
			writefile(CACHE_FILE, string.format('{"key":"%s","time":%d}', VALID_KEY, os.time()))
		end)
	end
end

local function hasValidCachedKey()
	if typeof(isfile) == "function" and typeof(readfile) == "function" then
		local ok, raw = pcall(function()
			if isfile(CACHE_FILE) then
				return readfile(CACHE_FILE)
			end
			return nil
		end)
		if ok and type(raw) == "string" then
			local savedKey = raw:match('"key"%s*:%s*"([^"]+)"')
			local savedTime = tonumber(raw:match('"time"%s*:%s*(%d+)'))
			if savedKey == VALID_KEY and savedTime and (os.time() - savedTime) < CACHE_DURATION then
				return true
			end
		end
	end
	return false
end

if hasValidCachedKey() then
	loadScript()
	return
end

local function setStatus(text, textColor, bannerColor)
	statusLabel.Text = text
	statusLabel.TextColor3 = textColor
	TweenService:Create(
		statusBanner,
		TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{ BackgroundColor3 = bannerColor }
	):Play()
end

local isShaking = false
local function smoothShakeMenu()
	if isShaking then return end
	isShaking = true

	local originPos = mainFrame.Position
	local wave = { 7, -7, 5.5, -5.5, 3.5, -3.5, 1.5, -1.5, 0 }

	TweenService:Create(
		inputHolderStroke,
		TweenInfo.new(0.15, Enum.EasingStyle.Sine, Enum.EasingDirection.Out),
		{ Color = Color3.fromRGB(255, 88, 102), Transparency = 0.25 }
	):Play()

	task.spawn(function()
		for i, offsetX in ipairs(wave) do
			local duration = 0.042 + (i * 0.002)
			local tween = TweenService:Create(
				mainFrame,
				TweenInfo.new(duration, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
				{
					Position = UDim2.new(
						originPos.X.Scale,
						originPos.X.Offset + offsetX,
						originPos.Y.Scale,
						originPos.Y.Offset
					)
				}
			)
			tween:Play()
			tween.Completed:Wait()
		end

		mainFrame.Position = originPos
		TweenService:Create(
			inputHolderStroke,
			TweenInfo.new(0.28, Enum.EasingStyle.Sine, Enum.EasingDirection.Out),
			{ Color = Color3.fromRGB(255, 255, 255), Transparency = 0.9 }
		):Play()
		isShaking = false
	end)
end

		end)
		setStatus("Key link copied to clipboard", Color3.fromRGB(115, 195, 255), Color3.fromRGB(22, 36, 56))
	else

		smoothShakeMenu()
	end

checkKeyBtn.MouseButton1Click:Connect(function()
	local cleanedKey = inputBox.Text:gsub("^%s*(.-)%s*$", "%1")

	if cleanedKey == "" then
		inputBox.Text = ""
		setStatus("Error: Please enter a key before checking", Color3.fromRGB(255, 105, 115), Color3.fromRGB(52, 22, 28))
		smoothShakeMenu()
		return
	end

	if cleanedKey == VALID_KEY then
		setStatus("Success: Key verified, launching KixDev...", Color3.fromRGB(95, 235, 145), Color3.fromRGB(20, 48, 34))
		saveKeyCache()
		task.wait(0.35)
		screenGui:Destroy()
		loadScript()
	else
		inputBox.Text = ""
		setStatus("Error: Invalid key! Please try again", Color3.fromRGB(255, 105, 115), Color3.fromRGB(52, 22, 28))
		smoothShakeMenu()
	end
end)
